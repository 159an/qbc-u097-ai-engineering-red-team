  import sys; sys.path.insert(0,'.')
  from services.solver.main import solve_direct
  from fastapi.responses import JSONResponse
  nodes=11; L=1.0; alpha=1.0; dx=L/(nodes-1)
  def run_v(v):
      p={'scheme':'btcs','alpha':alpha,'nodes':nodes,'dt':0.01,'tEnd':50.0,'length':L,
         'initial':{'kind':'sin','amplitude':1.0,'modes':1},
         'boundary':{'left':{'kind':'dirichlet','value':0.0},'right':{'kind':'dirichlet','value':1.0}},
         'advection':{'enabled':True,'velocity':float(v)},
         'probes':[round(i/(nodes-1),6) for i in range(1,nodes-1)],'recordEvery':10**6}
      r=solve_direct(p); assert not isinstance(r,JSONResponse), r
      pts=[q['points'][-1]['u'] for q in r['probes']]; return max(pts),min(pts)
  eps=1e-9
  for v in [0,5,10,15,18,19,20,21,22,24,26,30,40]:
      mx,mn=run_v(v); print(f'v={v:2d} Pe_cell={0.1*v:5.2f} u_max={mx:8.4f} u_min={mn:8.4f} oscillates={(mn<-eps) or (mx>1+eps)}')
  PY

结论（P3 Pe_cell 振荡阈值）：
  自变量 = v（nodes 固定 11 => dx=0.1 => Pe_cell = 0.1*v）；BTCS 无条件稳定，dt=0.01 到稳态
  彻底绕开 FTCS 的 CFL（r>0.5 会爆）。判据只用一条：解越过边界值域 [0,1]（u_min<-1e-9 或 u_max>1+1e-9）。
  v<=20（Pe_cell<=2.0）时 u_min>=0 不越界；v=21（Pe_cell=2.1）起 u_min<0 越界。
  二分（v∈[20,22]）收敛到 lo_v=hi_v=20.0，Pe_cell* = 0.1*20 = 2.000。
  ground-truth.json: P3_peclet_oscillation_threshold.measured = 2.0，theory = 2.0，落入 2±0.1 验收带。
